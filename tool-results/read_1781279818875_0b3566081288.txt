    84→    84→function getBearing(start: [number, number], end: [number, number]): number {
    85→    85→  const startLat = start[0] * Math.PI / 180;
    86→    86→  const endLat = end[0] * Math.PI / 180;
    87→    87→  const dLng = (end[1] - start[1]) * Math.PI / 180;
    88→    88→  const y = Math.sin(dLng) * Math.cos(endLat);
    89→    89→  const x = Math.cos(startLat) * Math.sin(endLat) - Math.sin(startLat) * Math.cos(endLat) * Math.cos(dLng);
    90→    90→  return (Math.atan2(y, x) * 180 / Math.PI + 360) % 360;
    91→    91→}

   103→   103→function LocationFinder({
   104→   104→  userLocationRef, setUserActualLocation, setShowMapPin, setArrowPos, setGpsStatus, hasInitialCenteredRef, userInteractedRef
   105→   105→}: {
   106→   106→  userLocationRef: React.MutableRefObject<[number, number] | null>;
   107→   107→  setUserActualLocation: (loc: [number, number] | null) => void;
   108→   108→  setShowMapPin: (val: boolean) => void;
   109→   109→  setArrowPos: (pos: { x: number; y: number; angle: number } | null) => void;
   110→   110→  setGpsStatus: (status: GpsStatus) => void;
   111→   111→  /** Ref shared with parent to track if the map has been initially centered */
   112→   112→  hasInitialCenteredRef: React.MutableRefObject<boolean>;
   113→   113→  /** Ref shared with parent to track if the user has manually interacted with the map */
   114→   114→  userInteractedRef: React.MutableRefObject<boolean>;
   115→   115→}) {
   116→   116→  const map = useMap();
   117→   117→
   118→   118→  useMapEvents({
   119→   119→    locationfound(e) {
   120→   120→      const { lat, lng } = e.latlng;
   121→   121→      const userCoords: [number, number] = [lat, lng];
   122→   122→      setUserActualLocation(userCoords);
   123→   123→      userLocationRef.current = userCoords;
   124→   124→      setGpsStatus("available");
   125→   125→
   126→   126→      const userLatLng = L.latLng(lat, lng);
   127→   127→
   128→   128→      // ALWAYS show the pin when location is found
   129→   129→      setShowMapPin(true);
   130→   130→
   131→   131→      // Only auto-center the map on the FIRST successful GPS acquisition.
   132→   132→      // After that, respect the user's manual navigation (drag/zoom/pan).
   133→   133→      // The user's position still updates in the background for GPS tracking,
   134→   134→      // vehicle markers, ETA logic, and the off-screen arrow indicator.
   135→   135→      if (!hasInitialCenteredRef.current) {
   136→   136→        // Fly to user if they are near the route (smooth animation, high zoom)
   137→   137→        if (routeBounds.contains(userLatLng)) {
   138→   138→          setArrowPos(null);
   139→   139→          map.flyTo([lat, lng], 16, { duration: 1.5 });
   140→   140→        }
   141→   141→        // If in the general map area but not on the route, use setView.
   142→   142→        else if (mapBounds.contains(userLatLng)) {
   143→   143→          setArrowPos(null);
   144→   144→          map.setView([lat, lng], 13, { animate: true });
   145→   145→        }
   146→   146→        // Mark that we've done the initial centering.
   147→   147→        // Even if the user was outside bounds (no camera move), we still mark it
   148→   148→        // so that subsequent GPS updates don't attempt to force-center.
   149→   149→        hasInitialCenteredRef.current = true;
   150→   150→      }
   151→   151→    },
   152→   152→    locationerror(e: any) {
   153→   153→      // Map GeolocationPositionError codes to GpsStatus
   154→   154→      // code 1 = PERMISSION_DENIED, code 2 = POSITION_UNAVAILABLE, code 3 = TIMEOUT
   155→   155→      const code = e?.code;
   156→   156→      if (code === 1) {
   157→   157→        setGpsStatus("denied");
   158→   158→      } else {
   159→   159→        setGpsStatus("unavailable");
   160→   160→      }
   161→   161→    },
   162→   162→    move() {
   163→   163→      const userCoords = userLocationRef.current;
   164→   164→      if (!userCoords) return;
   165→   165→
   166→   166→      const userLatLng = L.latLng(userCoords[0], userCoords[1]);
   167→   167→      const bounds = map.getBounds();
   168→   168→
   169→   169→      if (bounds.contains(userLatLng)) {
   170→   170→        setArrowPos(null);
   171→   171→        return;
   172→   172→      }
   173→   173→
   174→   174→      const center = map.getCenter();
   175→   175→      const angle = getBearing([center.lat, center.lng], userCoords);
   176→   176→
   177→   177→      const dx = Math.sin(angle * Math.PI / 180);
   178→   178→      const dy = -Math.cos(angle * Math.PI / 180);
   179→   179→
   180→   180→      const safeMinX = 5; const safeMaxX = 95;
   181→   181→      const safeMinY = 15; const safeMaxY = 75;
   182→   182→
   183→   183→      let tX = Infinity; let tY = Infinity;
   184→   184→
   185→   185→      if (dx > 0) tX = (safeMaxX - 50) / dx;
   186→   186→      else if (dx < 0) tX = (safeMinX - 50) / dx;
   187→   187→      if (dy > 0) tY = (safeMaxY - 50) / dy;
   188→   188→      else if (dy < 0) tY = (safeMinY - 50) / dy;
   189→   189→
   190→   190→      const t = Math.min(tX, tY);
   191→   191→
   192→   192→      setArrowPos({ x: 50 + (t * dx), y: 50 + (t * dy), angle: angle });
   193→   193→    },
   194→   194→    dragstart() {
   195→   195→      map.closePopup();
   196→   196→      // Mark that the user has manually interacted with the map.
   197→   197→      // This prevents any future auto-centering from overriding their position.
   198→   198→      userInteractedRef.current = true;
   199→   199→    },
   200→   200→    zoomstart() {
   201→   201→      // Zoom interactions also count as manual interaction.
   202→   202→      userInteractedRef.current = true;
   203→   203→    },
   204→   204→  });
   205→   205→
   206→   206→  useEffect(() => {
   207→   207→    // FIX: Increased timeout from 10000ms to 30000ms for mobile GPS cold starts.
   208→   208→    // Mobile GPS (especially iOS) can take 10-30+ seconds for a cold start,
   209→   209→    // while desktop browsers use instant WiFi-based location.
   210→   210→    // watch: true makes it continuously track the user's location instead of just asking once
   211→   211→    map.locate({ setView: false, maxZoom: 16, enableHighAccuracy: true, timeout: 30000, watch: true });
   212→   212→  }, [map]);
   213→   213→
   214→   214→  // FIX: Force Leaflet to recalculate container dimensions after mount.
   215→   215→  // On mobile, the bottom sheet and bottom nav overlap the map container,
   216→   216→  // and Leaflet may cache incorrect dimensions from before those elements
   217→   217→  // rendered. invalidateSize() forces a recalculation so map.locate()
   218→   218→  // and all coordinate calculations work correctly.
   219→   219→  useEffect(() => {
   220→   220→    const timer = setTimeout(() => {
   221→   221→      map.invalidateSize();
   222→   222→    }, 500);
   223→   223→    return () => clearTimeout(timer);
   224→   224→  }, [map]);
   225→   225→
   226→   226→  return null;
   227→   227→}

   230→   230→interface CommuterMapProps {

   237→   237→// Prevents excessive rerenders from high-frequency GPS updates.
   238→   238→// Spec requires 3–5 seconds; 3500ms ensures we stay within that range
   239→   239→// while allowing the 4s simulation tick to always trigger a recalculation.
   240→   240→const CALC_THROTTLE_MS = 3500;

   242→   242→// Vehicle simulation tick interval (ms).
   243→   243→// Each tick advances all vehicles one step along the route.
   244→   244→const SIMULATION_TICK_MS = 4000;

   246→   246→export default function CommuterMap({ isDesktop = false, onNearbyVehiclesChange }: CommuterMapProps) {

   251→   251→

   254→   254→

   257→   257→

   260→   260→

   262→   262→  const lastCalcRef = useRef(0);
   263→   263→

   265→   265→  // After the first successful locationfound → flyTo/setView, this becomes true
   266→   266→  // and subsequent GPS updates will NOT re-center the map viewport.
   267→   267→  const hasInitialCenteredRef = useRef(false);

   269→   269→  // Track whether the user has manually interacted with the map (drag/zoom).
   270→   270→  // This is an additional safety layer — even if initial centering hasn't happened
   271→   271→  // yet, user interaction takes priority and prevents auto-centering.
   272→   272→  const userInteractedRef = useRef(false);

   279→   279→  // --- Cleanup: clear notification dedup set on unmount ---
   280→   280→  // WHY: The proximity-notification module keeps a Set of notified vehicle plates
   281→   281→  // to avoid spamming the same notification. Without clearing on unmount, this Set
   282→   282→  // grows indefinitely during a session. If the user navigates away and back, they

   284→   284→  // Clearing on unmount ensures a fresh start each time the map mounts.

   291→   291→  // --- Vehicle simulation interval ---

   293→   293→  // This interval advances all vehicles along the route every 4 seconds.
   294→   294→  useEffect(() => {

   304→   304→  // The simulation tick (4s) drives most updates; GPS updates between
   305→   305→  // ticks are throttled away.

   308→   308→

   320→   320→        capacityStatus: v.capacity === "AVAILABLE" ? "Available" : v.capacity === "STANDING" ? "Standing" : "Full",
   321→   321→        speed: 25,

   323→   323→      };

   327→   327→      vehicleLocations,
   328→   328→      userActualLocation[0],
   329→   329→      userActualLocation[1]
   330→   330→    );
   331→   331→
   332→   332→    setRadiusResult(result);
   333→   333→

   335→   335→    result.withinRadius.forEach((v) => {
   336→   336→      sendProximityNotification(v.plateNumber, formatDistance(v.distanceInMeters));
   337→   337→    });
   338→   338→
   339→   339→    // Clear notification dedup when commuter exits ALL conductor radii.
   340→   340→    // WHY: If the commuter walks out of range and back in, they should
   341→   341→    // receive fresh notifications. Without clearing, the Set would still
   342→   342→    // contain the old plate numbers and suppress the re-notification.
   343→   343→    if (result.withinRadius.length === 0) {
   344→   344→      clearNotifiedVehicles();
   345→   345→    }
   346→   346→  }, [userActualLocation, activeVehicles]);
   347→   347→
   348→   348→  // --- Notify parent of tracking updates ---
   349→   349→  // WHY: Dashboard needs both the within-radius vehicles AND the GPS status
   350→   350→  // to correctly render ETA panel, hail button state, and status messages.
   351→   351→  // Separated from the computation effect to keep concerns clean.
   352→   352→  useEffect(() => {
   353→   353→    if (gpsStatus === "available") {
   354→   354→      onNearbyVehiclesChange?.(radiusResult?.withinRadius || [], gpsStatus);
   355→   355→    } else {
   356→   356→      // When GPS is not available, send empty vehicles with current status
   357→   357→      // so the dashboard can show the appropriate GPS error state
   358→   358→      onNearbyVehiclesChange?.([], gpsStatus);
   359→   359→    }
   360→   360→  }, [radiusResult, gpsStatus, onNearbyVehiclesChange]);
