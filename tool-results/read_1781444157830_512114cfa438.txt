    83→  status: AsyncStatus;
    84→  data: T;
    85→  error: string | null;
    86→}
    87→
    88→export function loadingState<T>(fallback: T): AsyncState<T> {
    89→  return { status: "loading", data: fallback, error: null };
    90→}

    92→export function successState<T>(data: T): AsyncState<T> {
    93→  return { status: "success", data, error: null };
    94→}
    95→

    97→  return { status: "error", data: fallback, error };

   326→========== /tmp/chatco-repo/lib/api/client.ts ==========
   327→/**
   328→ * API Client — Centralized fetch wrapper for Laravel + Supabase backend.
   329→ *
   330→ * Architecture:
   331→ * ─────────────────────────────────────────────────────────────
   332→ * - Laravel serves as the API gateway (sanctum auth, business logic)
   333→ * - Supabase (Postgres) is the database behind Laravel's Eloquent models
   334→ * - All API calls go through this client for consistent error handling,
   335→ *   auth headers, and base URL configuration
   336→ *
   337→ * Usage:
   338→ *   import { api } from "@/lib/api/client";
   339→ *   const data = await api.get<CommuterProfile>("/api/commuter/profile");
   340→ *   const result = await api.post<Transaction>("/api/transactions", payload);
   341→ */
   342→
   343→// ─── Configuration ───────────────────────────────────────────────────
   344→
   345→const API_BASE_URL = process.env.NEXT_PUBLIC_API_URL || "";
   346→
   347→interface ApiConfig {
   348→  baseUrl: string;
   349→  credentials: RequestCredentials;
   350→  defaultHeaders: Record<string, string>;
   351→}
   352→
   353→const defaultConfig: ApiConfig = {
   354→  baseUrl: API_BASE_URL,
   355→  credentials: "include", // Sanctum uses cookie-based auth
   356→  defaultHeaders: {
   357→    "Content-Type": "application/json",
   358→    Accept: "application/json",
   359→  },
   360→};
   361→
   362→// ─── Error Types ─────────────────────────────────────────────────────
   363→
   364→export class ApiError extends Error {
   365→  constructor(
   366→    public status: number,
   367→    public statusText: string,
   368→    public body: unknown
   369→  ) {
   370→    super(`API Error ${status}: ${statusText}`);
   371→    this.name = "ApiError";
   372→  }
   373→}
   374→
   375→export class NetworkError extends Error {
   376→  constructor(message: string) {
   377→    super(message);
   378→    this.name = "NetworkError";
   379→  }
   380→}
   381→
   382→// ─── Response Types ──────────────────────────────────────────────────
   383→
   384→export interface ApiResponse<T> {
   385→  data: T;
   386→  meta?: {
   387→    page: number;
   388→    perPage: number;
   389→    total: number;
   390→    totalPages: number;
   391→  };
   392→}
   393→
   394→// ─── Client ──────────────────────────────────────────────────────────
   395→
   396→class ApiClient {
   397→  private config: ApiConfig;
   398→
   399→  constructor(config: ApiConfig = defaultConfig) {
   400→    this.config = config;
   401→  }
   402→
   403→  private async request<T>(
   404→    path: string,
   405→    options: RequestInit = {}
   406→  ): Promise<T> {
   407→    const url = `${this.config.baseUrl}${path}`;
   408→
   409→    const headers = {
   410→      ...this.config.defaultHeaders,
   411→      ...options.headers,
   412→    };
   413→
   414→    try {
   415→      const response = await fetch(url, {
   416→        ...options,
   417→        headers,
   418→        credentials: this.config.credentials,
   419→      });
   420→
   421→      // Handle non-JSON responses (e.g., 204 No Content)
   422→      if (response.status === 204) {
   423→        return undefined as T;
   424→      }
   425→
   426→      const data = await response.json();
   427→
   428→      if (!response.ok) {
   429→        throw new ApiError(response.status, response.statusText, data);
   430→      }
   431→
   432→      return data as T;
   433→    } catch (error) {
   434→      if (error instanceof ApiError) throw error;
   435→      throw new NetworkError(
   436→        error instanceof Error ? error.message : "Network request failed"
   437→      );
   438→    }
   439→  }
   440→
   441→  async get<T>(path: string, params?: Record<string, string>): Promise<T> {
   442→    const searchParams = params
   443→      ? `?${new URLSearchParams(params).toString()}`
   444→      : "";
   445→    return this.request<T>(`${path}${searchParams}`);
   446→  }
   447→
   448→  async post<T>(path: string, body?: unknown): Promise<T> {
   449→    return this.request<T>(path, {
   450→      method: "POST",
   451→      body: body ? JSON.stringify(body) : undefined,
   452→    });
   453→  }
   454→
   455→  async put<T>(path: string, body?: unknown): Promise<T> {
   456→    return this.request<T>(path, {
   457→      method: "PUT",
   458→      body: body ? JSON.stringify(body) : undefined,
   459→    });
   460→  }
   461→
   462→  async patch<T>(path: string, body?: unknown): Promise<T> {
   463→    return this.request<T>(path, {
   464→      method: "PATCH",
   465→      body: body ? JSON.stringify(body) : undefined,
   466→    });
   467→  }
   468→
   469→  async delete<T>(path: string): Promise<T> {
   470→    return this.request<T>(path, { method: "DELETE" });
   471→  }
   472→}
