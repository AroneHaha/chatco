    31→
    32→interface FareCalcModalProps {
    33→  isOpen: boolean;

    40→
    41→/**
    42→ * Step flow:
    43→ * - Cash:  method → select (with commuter type + fare overview) → confirm → processing → success/failed
    44→ * - GCash: method → select (locations only, NO fare) → qr_code → scan_result (fare breakdown after scan) → confirm → processing → success/failed
    45→ */
    46→type Step = "method" | "select" | "confirm" | "processing" | "qr_code" | "scan_result" | "success" | "failed";
    47→type SelectedPaymentMethod = "GCash" | "Cash";
    48→
    49→export default function FareCalcModal({ isOpen, onClose, shiftId, conductorName, unitNumber, driverName }: FareCalcModalProps) {
    50→  const [step, setStep] = useState<Step>("method");
    51→  const [selectedMethod, setSelectedMethod] = useState<SelectedPaymentMethod | null>(null);
    52→  const [pickupPoint, setPickupPoint] = useState<PointArea | null>(null);
    53→  const [dropoffPoint, setDropoffPoint] = useState<PointArea | null>(null);
    54→  const [commuterType, setCommuterType] = useState<CommuterType>("REGULAR");
    55→  const [searchQuery, setSearchQuery] = useState("");
    56→  const [selectingField, setSelectingField] = useState<
    57→    "pickup" | "dropoff" | null
    58→  >("pickup");
    59→  const [expandedBarangay, setExpandedBarangay] = useState<number | null>(null);
    60→  const [pickupLandmark, setPickupLandmark] = useState<string | null>(null);
    61→  const [dropoffLandmark, setDropoffLandmark] = useState<string | null>(null);
    62→  const [paymentIntent, setPaymentIntent] =
    63→    useState<GCashPaymentIntent | null>(null);
    64→  const [paymentStatus, setPaymentStatus] = useState<PaymentStatus | null>(
    65→    null
    66→  );
    67→  // GCash scan result state
    68→  const [chargeResult, setChargeResult] = useState<ChargeFareResult | null>(null);
    69→  const [scannedCommuterType, setScannedCommuterType] = useState<CommuterType>("REGULAR");
    70→  const [scannedCommuterName, setScannedCommuterName] = useState("Commuter");
    71→  const [scannedCommuterId, setScannedCommuterId] = useState("c_001");
    72→
    73→  // ─── Fare Calculation using FARE_MATRIX (correct data source) ───
    74→  const fareInfo = useMemo(() => {
    75→    if (!pickupPoint || !dropoffPoint) return null;
    76→
    77→    const isDiscounted = commuterType !== "REGULAR";
    78→    const barangaysTraveled = getBarangaysTraversed(
    79→      pickupPoint.pointNumber,
    80→      dropoffPoint.pointNumber
    81→    );
    82→    const regularFare = getFareBetween(
    83→      pickupPoint.pointNumber,
    84→      dropoffPoint.pointNumber,
    85→      false
    86→    );
    87→    const discountedFare = getFareBetween(
    88→      pickupPoint.pointNumber,
    89→      dropoffPoint.pointNumber,
    90→      true
    91→    );
    92→    const finalFare = isDiscounted ? discountedFare : regularFare;
    93→    const discountAmount = regularFare - discountedFare;
    94→    const succeedingCount = Math.max(
    95→      0,
    96→      barangaysTraveled - FARE_CONFIG.BASE_BARANGAY_COUNT
    97→    );
    98→
    99→    return {
   100→      barangaysTraveled,
   101→      regularFare,
   102→      discountedFare,
   103→      finalFare,
   104→      hasDiscount: isDiscounted,
   105→      discountAmount,
   106→      succeedingCount,
   107→      baseBarangayCount: FARE_CONFIG.BASE_BARANGAY_COUNT,
   108→    };
   109→  }, [pickupPoint, dropoffPoint, commuterType]);
   110→
   111→  // ─── Fare calculation for GCash after scan (uses scanned commuter type) ───
   112→  const gcashFareInfo = useMemo(() => {
   113→    if (!pickupPoint || !dropoffPoint) return null;
   114→
   115→    const isDiscounted = scannedCommuterType !== "REGULAR";
   116→    const barangaysTraveled = getBarangaysTraversed(
   117→      pickupPoint.pointNumber,
   118→      dropoffPoint.pointNumber
   119→    );
   120→    const regularFare = getFareBetween(
   121→      pickupPoint.pointNumber,
   122→      dropoffPoint.pointNumber,
   123→      false
   124→    );
   125→    const discountedFare = getFareBetween(
   126→      pickupPoint.pointNumber,
   127→      dropoffPoint.pointNumber,
   128→      true
   129→    );
   130→    const finalFare = isDiscounted ? discountedFare : regularFare;
   131→    const discountAmount = regularFare - discountedFare;
   132→    const succeedingCount = Math.max(
   133→      0,
   134→      barangaysTraveled - FARE_CONFIG.BASE_BARANGAY_COUNT
   135→    );
   136→
   137→    return {
   138→      barangaysTraveled,
   139→      regularFare,
   140→      discountedFare,
   141→      finalFare,
   142→      hasDiscount: isDiscounted,
   143→      discountAmount,
   144→      succeedingCount,
   145→      baseBarangayCount: FARE_CONFIG.BASE_BARANGAY_COUNT,
   146→    };
   147→  }, [pickupPoint, dropoffPoint, scannedCommuterType]);
   148→
   149→  const filteredPoints = searchQuery
   150→    ? FARE_MATRIX.filter(
   151→        (p) =>
   152→          p.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
   153→          p.code.toLowerCase().includes(searchQuery.toLowerCase()) ||
   154→          p.landmarks.some((l) =>
   155→            l.toLowerCase().includes(searchQuery.toLowerCase())
   156→          )
   157→      )
   158→    : FARE_MATRIX;
   159→

   162→    if (!pickupPoint || !dropoffPoint || !shiftId) return;
   163→
   164→    const fareData = overrideFare || fareInfo;
   165→    if (!fareData) return;
   166→
   167→    const paymentMethodType: PaymentMethodType = method === "GCash" ? "GCash_Scanned" : "Cash";
   168→    const effectiveCommuterType = overrideCommuterType || commuterType;
   169→
   170→    await createTransaction(shiftId, {
   171→      paymentMethod: paymentMethodType,
   172→      finalAmount: fareData.finalFare,
   173→      passengerName: method === "GCash" ? scannedCommuterName : "Commuter",
   174→      passengerId: method === "GCash" ? scannedCommuterId : "c_001",
   175→      passengerRole: effectiveCommuterType,
   176→      from: pickupPoint.name,
   177→      to: dropoffPoint.name,
   178→      distance: fareData.barangaysTraveled,
   179→      baseFare: fareData.regularFare,
   180→      succeedingKm: fareData.succeedingCount,
   181→      discountAmount: fareData.discountAmount,
   182→      conductorName: conductorName || "—",
   183→      unitNumber: unitNumber || "—",
   184→      driverName: driverName || "—",
   185→    });
   186→  };
   187→
   188→  const handlePayWithGCash = async () => {
   189→    if (!gcashFareInfo || !pickupPoint || !dropoffPoint) return;
   190→
   191→    setStep("processing");
   192→
   193→    try {
   194→      // Use gcashService.chargeFare() which internally calls /api/gcash/charge
   195→      // with dev fallback for when the backend is unavailable
   196→      const result = await chargeFare({
   197→        amount: gcashFareInfo.finalFare,
   198→        commuterId: scannedCommuterId,
   199→        commuterName: scannedCommuterName,
   200→        commuterType: scannedCommuterType,
   201→        pickupPoint: pickupPoint.pointNumber,
   202→        dropoffPoint: dropoffPoint.pointNumber,
   203→        conductorId: conductorName,
   204→        shiftId,
   205→        unitNumber: unitNumber,
   206→      });
   207→
   208→      setChargeResult(result);
   209→
   210→      if (result.success) {
   211→        await recordTransaction("GCash", gcashFareInfo, scannedCommuterType);
   212→        setStep("success");
   213→      } else {
   214→        setStep("failed");
   215→      }
   216→    } catch {
   217→      setStep("failed");
   218→    }
   219→  };
   220→
   221→  const handlePayWithCash = async () => {
   222→    if (!fareInfo || !pickupPoint || !dropoffPoint) return;
   223→
   224→    setStep("processing");
   225→
   226→    // Cash payment: brief processing (just recording), then success
   227→    await new Promise((r) => setTimeout(r, 800));
   228→
   229→    await recordTransaction("Cash");
   230→    setStep("success");
   231→  };
   232→
   233→  const handleConfirmPayment = () => {
   234→    if (selectedMethod === "GCash") {
   235→      handlePayWithGCash();
   236→    } else {
   237→      handlePayWithCash();
   238→    }
   239→  };
   240→
   241→  // ─── Handle locations selected in GCash mode ────────────────────
   242→  // After selecting pickup & dropoff for GCash, go directly to QR code
   243→  // (no fare display at this step)
   244→  const handleGCashLocationsSelected = () => {
   245→    setStep("qr_code");
   246→  };
   247→
   248→  // ─── Simulate commuter QR scan ──────────────────────────────────
   249→  // In production, the commuter scans the conductor's QR code and the
   250→  // system detects the commuter type from their e-chatco account.

   262→    const detectedId = "c_001";
   263→

   265→    setScannedCommuterName(detectedName);

   270→

   276→  const pickupColor = isSameBarangay ? "violet" : "green";

   278→

   280→  const pickupDotClass = isSameBarangay ? "bg-violet-500" : "bg-emerald-500";
   281→  const dropoffDotClass = isSameBarangay ? "bg-violet-500" : "bg-emerald-500";
   282→  const pickupBorderClass = isSameBarangay ? "border-violet-500/50 bg-violet-500/10" : "border-emerald-500/50 bg-emerald-500/10";
   283→  const dropoffBorderClass = isSameBarangay ? "border-violet-500/50 bg-violet-500/10" : "border-emerald-500/50 bg-emerald-500/10";
   284→  const pickupBadgeClass = isSameBarangay ? "text-violet-400 bg-violet-500/10" : "text-emerald-400 bg-emerald-500/10";
   285→  const dropoffBadgeClass = isSameBarangay ? "text-violet-400 bg-violet-500/10" : "text-emerald-400 bg-emerald-500/10";
   286→  const pickupTagClass = isSameBarangay ? "bg-violet-500 text-white" : "bg-emerald-500 text-white";
   287→  const dropoffTagClass = isSameBarangay ? "bg-violet-500 text-white" : "bg-emerald-500 text-white";

   290→
   291→  // ─── Clear location handlers (only reset location, NOT payment) ──
   292→  const clearPickup = () => {

   294→    setPickupLandmark(null);
   295→    setSelectingField("pickup");
   296→    // Collapse expanded barangay if it belonged to the cleared pickup
   297→    if (expandedBarangay === pickupPoint?.pointNumber) {
   298→      setExpandedBarangay(null);
   299→    }
   300→  };
   301→
   302→  const clearDropoff = () => {
   303→    setDropoffPoint(null);
   304→    setDropoffLandmark(null);
   305→    setSelectingField("dropoff");
   306→    // Collapse expanded barangay if it belonged to the cleared dropoff
   307→    if (expandedBarangay === dropoffPoint?.pointNumber) {
   308→      setExpandedBarangay(null);
   309→    }
   310→  };
   311→
   312→  // ─── RESET STATE & CLOSE ──────────────────────────────────────

   331→  };
   332→
   333→  // ─── Auto-advance from QR code to simulate scan after 1s ──────
   334→  // In production, this would wait for a real-time event from the

   339→    return () => clearTimeout(timer);
   340→  }, [step]);
   341→
   342→  // ─── Auto-expand landmark when switching selectingField ────────
   343→  // When the user clicks the pickup/dropoff card to re-select,

   346→  useEffect(() => {
   347→    if (step !== "select") return;
   348→    if (selectingField === "pickup" && pickupPoint && pickupPoint.landmarks.length > 0) {
   349→      setExpandedBarangay(pickupPoint.pointNumber);
   350→    } else if (selectingField === "dropoff" && dropoffPoint && dropoffPoint.landmarks.length > 0) {
   351→      setExpandedBarangay(dropoffPoint.pointNumber);
   352→    }
   353→  }, [selectingField, step, pickupPoint, dropoffPoint]);
   354→
   355→  // ─── HIDDEN UNTIL CLICKED ──────────────────────────────────────
   356→  if (!isOpen) return null;
   357→
   358→  // ─── STEP: Payment Method Selection ────────────────────────────

   360→  if (step === "method") {

   382→            {/* Cash Option */}

   384→              onClick={() => {
   385→                setSelectedMethod("Cash");

   387→              }}
   388→              className="w-full text-left p-4 rounded-xl border border-white/10 bg-white/5 hover:bg-emerald-500/10 hover:border-emerald-500/30 transition-all duration-200 group"
   389→            >
   390→              <div className="flex items-center gap-4">
   391→                <div className="w-12 h-12 rounded-xl bg-emerald-500/15 border border-emerald-500/20 flex items-center justify-center flex-shrink-0 group-hover:bg-emerald-500/25 transition-colors">
   392→                  <svg className="w-6 h-6 text-emerald-400" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={1.8}>
   393→                    <path strokeLinecap="round" strokeLinejoin="round" d="M2.25 18.75a60.07 60.07 0 0 1 15.797 2.101c.727.198 1.453-.342 1.453-1.096V18.75M3.75 4.5v.75A.75.75 0 0 1 3 6h-.75m0 0v-.375c0-.621.504-1.125 1.125-1.125H20.25M2.25 6v9m18-10.5v.75c0 .414.336.75.75.75h.75m-1.5-1.5h.375c.621 0 1.125.504 1.125 1.125v9.75c0 .621-.504 1.125-1.125 1.125h-.375m1.5-1.5H21a.75.75 0 0 0-.75.75v.75m0 0H3.75m0 0h-.375a1.125 1.125 0 0 1-1.125-1.125V15m1.5 1.5v-.75A.75.75 0 0 0 3 15h-.75M15 10.5a3 3 0 1 1-6 0 3 3 0 0 1 6 0Zm3 0h.008v.008H18V10.5Zm-12 0h.008v.008H6V10.5Z" />

   405→
   406→            {/* GCash Option */}
   407→            <button
   408→              onClick={() => {
   409→                setSelectedMethod("GCash");
   410→                setStep("select");
   411→              }}
   412→              className="w-full text-left p-4 rounded-xl border border-white/10 bg-white/5 hover:bg-blue-500/10 hover:border-blue-500/30 transition-all duration-200 group"
   413→            >
   414→              <div className="flex items-center gap-4">
   415→                <div className="w-12 h-12 rounded-xl bg-blue-500/15 border border-blue-500/20 flex items-center justify-center flex-shrink-0 group-hover:bg-blue-500/25 transition-colors">
   416→                  <svg className="w-6 h-6 text-blue-400" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={1.8}>
   417→                    <path strokeLinecap="round" strokeLinejoin="round" d="M10.5 1.5H8.25A2.25 2.25 0 0 0 6 3.75v16.5a2.25 2.25 0 0 0 2.25 2.25h7.5A2.25 2.25 0 0 0 18 20.25V3.75a2.25 2.25 0 0 0-2.25-2.25H13.5m-3 0V3h3V1.5m-3 0h3m-3 18.75h3" />
   418→                  </svg>
   419→                </div>

   441→
   442→  // ─── STEP: Point Area Selection (FULLSCREEN LOCATION PICKER) ──────
   443→
   444→  if (step === "select") {
   445→    // For GCash: no fare display, no commuter type selector

   457→              <div className="flex items-center gap-2 min-w-0">
   458→                <button
   459→                  onClick={() => setStep("method")}
   460→                  className="w-9 h-9 rounded-xl bg-white/5 border border-white/10 flex items-center justify-center text-white/50 hover:text-white hover:bg-white/10 transition-colors flex-shrink-0"
   461→                >
   462→                  <svg className="w-4 h-4" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2}>
