     1→// lib/fare-matrix-data.ts
     2→// Single source of truth for the CHATCO fare matrix.
     3→// 34 operational point areas along the Calumpit–Meycauayan jeepney route.
     4→//
     5→// FARE LOGIC:
     6→//   • Barangay-based fare progression model
     7→//   • The first few barangays traveled are covered by the BASE FARE
     8→//   • Every succeeding barangay beyond the base zone is one fare increment
     9→//   • Applies regardless of route direction
    10→//   • Bold items = BARANGAYS (fare zones); indented items = landmarks within the zone
    11→
    12→// ─── Types ───────────────────────────────────────────────────────────────────
    13→
    14→export interface FarePoint {
    15→  /** Sequential point number 1–34 along the route */
    16→  pointNumber: number;
    17→  /** Display name (includes landmark / barangay info) */
    18→  name: string;
    19→  /** Regular (full) fare – cumulative from Calumpit terminal (Point 34) */
    20→  regularFare: number;
    21→  /** Discounted fare (student / senior / PWD) – cumulative from Calumpit terminal */
    22→  discountedFare: number;
    23→  /** Landmark reference points within the same fare zone (same fare) */
    24→  subStops?: string[];
    25→}
    26→
    27→export interface FareResult {
    28→  fromPoint: FarePoint;
    29→  toPoint: FarePoint;
    30→  /** Number of point-area intervals between the two points */
    31→  barangaysTraveled: number;
    32→  /** Computed regular fare for this pair */
    33→  regularFare: number;
    34→  /** Computed discounted fare for this pair */
    35→  discountedFare: number;
    36→  /** Base fare that covers the first zone */
    37→  baseFare: number;
    38→  /** Per-barangay increment after the base zone */
    39→  succeedingFare: number;
    40→  /** How many succeeding increments applied */
    41→  succeedingCount: number;
    42→}
    43→
    44→// ─── Constants ───────────────────────────────────────────────────────────────
    45→
    46→/** Minimum regular fare (within base-fare zone — first 4 barangays) */
    47→const MINIMUM_REGULAR_FARE = 18.0;
    48→/** Minimum discounted fare (within base-fare zone — first 4 barangays) */
    49→const MINIMUM_DISCOUNTED_FARE = 14.4;
    50→
    51→/**
    52→ * The base fare (₱18) covers the first 4 barangays traveled.
    53→ * Every succeeding barangay beyond the 4th is one fare increment (₱2).
    54→ * Direction does not matter — same logic applies.
    55→ */
    56→const BASE_ZONE_INTERVALS = 4;
    57→
    58→/** Per-barangay fare increment after the base zone */
    59→const SUCCEEDING_FARE_INCREMENT = 2.0;

   271→// ─── Helpers ─────────────────────────────────────────────────────────────────
   272→
   273→function findPoint(pointNumber: number): FarePoint | undefined {
   274→  return initialFarePoints.find((p) => p.pointNumber === pointNumber);
   275→}
   276→
   277→export function getFareByPoints(
   278→  fromPointNumber: number,
   279→  toPointNumber: number
   280→): FareResult {
   281→  const fromPoint = findPoint(fromPointNumber);
   282→  const toPoint = findPoint(toPointNumber);
   283→
   284→  if (!fromPoint || !toPoint) {
   285→    throw new Error(
   286→      `Invalid point number: from=${fromPointNumber}, to=${toPointNumber}. ` +
   287→        `Valid range is 1–${initialFarePoints.length}.`
   288→    );
   289→  }
   290→
   291→  const barangaysTraveled = Math.abs(fromPointNumber - toPointNumber);
   292→
   293→  if (barangaysTraveled === 0) {
   294→    return {
   295→      fromPoint,
   296→      toPoint,
   297→      barangaysTraveled: 0,
   298→      regularFare: MINIMUM_REGULAR_FARE,
   299→      discountedFare: MINIMUM_DISCOUNTED_FARE,
   300→      baseFare: MINIMUM_REGULAR_FARE,
   301→      succeedingFare: SUCCEEDING_FARE_INCREMENT,
   302→      succeedingCount: 0,
   303→    };
   304→  }
   305→
   306→  const rawRegularFare = Math.abs(fromPoint.regularFare - toPoint.regularFare);
   307→  const rawDiscountedFare = Math.abs(
   308→    fromPoint.discountedFare - toPoint.discountedFare
   309→  );
   310→
   311→  const regularFare = Math.max(rawRegularFare, MINIMUM_REGULAR_FARE);
   312→  const discountedFare = Math.max(rawDiscountedFare, MINIMUM_DISCOUNTED_FARE);
   313→
   314→  const succeedingCount = Math.max(0, barangaysTraveled - BASE_ZONE_INTERVALS);
   315→
   316→  return {
   317→    fromPoint,
   318→    toPoint,
   319→    barangaysTraveled,
   320→    regularFare: Math.round(regularFare * 100) / 100,
   321→    discountedFare: Math.round(discountedFare * 100) / 100,
   322→    baseFare: MINIMUM_REGULAR_FARE,
   323→    succeedingFare: SUCCEEDING_FARE_INCREMENT,
   324→    succeedingCount,
   325→  };
   326→}

   328→// Single source of truth for CHATCO fare computation
   329→// Based on official CHATCO TSC fare matrix (Calumpit–Meycauayan route)
   330→//
   331→// OPERATIONAL LOGIC:
   332→// - BOLD items in the fare matrix = BARANGAYS (operational fare zones)
   333→// - Indented items under each barangay = landmarks, establishments, terminals,
   334→//   pickup points, or key locations within that barangay
   335→// - Fare computation is NOT based on exact GPS distance or unrestricted
   336→//   point-to-point map calculation
   337→// - The system follows a BARANGAY-BASED FARE PROGRESSION MODEL:
   338→//   * The first four (4) barangays traveled are covered by the base fare
   339→//   * Every succeeding barangay after the fourth is treated as one (1)
   340→//     succeeding kilometer or fare increment
   341→//   * This applies regardless of route direction or starting point
   342→//
   343→// FARE LOOKUP METHOD:
   344→// The fare between any two barangays depends ONLY on the NUMBER of barangays
   345→// traversed (not on which specific barangays). A lookup table maps the
   346→// barangay count to the actual fare amount, derived directly from the
   347→// cooperative's official fare matrix.
   348→
   349→export interface PointArea {
   350→  pointNumber: number;
   351→  code: string;
   352→  name: string; // Barangay name (the bold text in the fare matrix)
   353→  landmarks: string[]; // Landmarks/establishments within this barangay
   354→  regularFare: number; // Fare from Calumpit terminal (regular)
   355→  discountedFare: number; // Fare from Calumpit terminal (student/senior/PWD)
   356→  coordinates: { lat: number; lng: number };
   357→}
   358→

   780→/**
   781→ * Calculate fare between any two barangays using the barangay-based
   782→ * progression model.
   783→ *
   784→ * The fare depends ONLY on the number of barangays traversed:
   785→ *   barangaysTraversed = |destPosition - originPosition| + 1
   786→ *
   787→ * This value is then looked up in the official fare table.
   788→ *
   789→ * Example:
   790→ *   Origin: Calumpit (position 1)
   791→ *   Destination: Longos (position 6)
   792→ *   Barangays traversed: |6 - 1| + 1 = 6
   793→ *   Regular fare: ₱16.25 (from FARE_BY_BARANGAY_COUNT table)
   794→ *
   795→ * The same fare of ₱16.25 applies for ANY 6-barangay trip regardless
   796→ * of starting point (e.g., Borneo → CEU is also 6 barangays = ₱16.25).
   797→ */
   798→export function getFareBetween(
   799→  originPosition: number,
   800→  destinationPosition: number,
   801→  isDiscounted: boolean
   802→): number {
   803→  const origin = getPointByNumber(originPosition);
   804→  const destination = getPointByNumber(destinationPosition);
   805→
   806→  if (!origin || !destination) return 0;
   807→
   808→  const barangaysTraversed =
   809→    Math.abs(destination.pointNumber - origin.pointNumber) + 1;
   810→
   811→  const fareTable = isDiscounted
   812→    ? FARE_BY_BARANGAY_COUNT_DISCOUNTED
   813→    : FARE_BY_BARANGAY_COUNT_REGULAR;
   814→
   815→  return fareTable[barangaysTraversed] ?? (isDiscounted ? 12.0 : 15.0);
   816→}
   817→
   818→/**
   819→ * Get the number of barangays traversed between two points.
   820→ */
   821→export function getBarangaysTraversed(
   822→  originPosition: number,
   823→  destinationPosition: number
   824→): number {
   825→  return Math.abs(destinationPosition - originPosition) + 1;
   826→}

   828→/**
   829→ * Get the fare explanation string for a given trip.
   830→ * Useful for displaying fare breakdowns in the UI.
   831→ *
   832→ * Example output:
   833→ *   "Base fare covers first 4 barangays + 2 succeeding barangays"
   834→ *   "Base fare (within first 4 barangays)"
   835→ */
   836→export function getFareExplanation(
   837→  originPosition: number,
   838→  destinationPosition: number
   839→): string {
   840→  const count = getBarangaysTraversed(originPosition, destinationPosition);
   841→
   842→  if (count <= FARE_CONFIG.BASE_BARANGAY_COUNT) {
   843→    return `Base fare (within first ${count} barangay${count > 1 ? "s" : ""})`;
   844→  }
   845→
   846→  const succeeding = count - FARE_CONFIG.BASE_BARANGAY_COUNT;
   847→  return `Base fare covers first ${FARE_CONFIG.BASE_BARANGAY_COUNT} barangays + ${succeeding} succeeding barangay${succeeding > 1 ? "s" : ""}`;
   848→}

   850→/**
   851→ * Find the nearest operational barangay to a given GPS coordinate.
   852→ * Uses the Haversine formula for accurate distance calculation.
   853→ */
   854→export function findNearestPoint(lat: number, lng: number): PointArea {
   855→  let nearest = FARE_MATRIX[0];
   856→  let minDistance = Infinity;
   857→
   858→  for (const point of FARE_MATRIX) {
   859→    const distance = haversineDistance(
   860→      lat,
   861→      lng,
   862→      point.coordinates.lat,
   863→      point.coordinates.lng
   864→    );
   865→    if (distance < minDistance) {
   866→      minDistance = distance;
   867→      nearest = point;
   868→    }
   869→  }
   870→
   871→  return nearest;
   872→}

   875→ * Search barangays by name or landmark.
   876→ * Used for the pickup/dropoff selector UI.
   877→ */
   878→export function searchPoints(query: string): PointArea[] {
   879→  const q = query.toLowerCase().trim();
   880→  if (!q) return FARE_MATRIX;
   881→
   882→  return FARE_MATRIX.filter(
   883→    (p) =>
   884→      p.name.toLowerCase().includes(q) ||
   885→      p.code.toLowerCase().includes(q) ||
   886→      p.landmarks.some((l) => l.toLowerCase().includes(q))
   887→  );
   888→}

   904→/** Haversine distance in meters between two lat/lng points */
   905→function haversineDistance(
   906→  lat1: number,
   907→  lng1: number,
   908→  lat2: number,
   909→  lng2: number
   910→): number {
   911→  const R = 6_371_000; // Earth radius in meters
   912→  const toRad = (deg: number) => (deg * Math.PI) / 180;
   913→
   914→  const dLat = toRad(lat2 - lat1);
   915→  const dLng = toRad(lng2 - lng1);
   916→
   917→  const a =
   918→    Math.sin(dLat / 2) ** 2 +
   919→    Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLng / 2) ** 2;
   920→
   921→  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
   922→  return R * c;
   923→}
