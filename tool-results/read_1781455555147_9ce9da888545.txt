   109→   109→  }, [pickupPoint, dropoffPoint, commuterType]);
   110→   110→

   112→   112→  const gcashFareInfo = useMemo(() => {
   113→   113→    if (!pickupPoint || !dropoffPoint) return null;
   114→   114→
   115→   115→    const isDiscounted = scannedCommuterType !== "REGULAR";
   116→   116→    const barangaysTraveled = getBarangaysTraversed(
   117→   117→      pickupPoint.pointNumber,
   118→   118→      dropoffPoint.pointNumber
   119→   119→    );
   120→   120→    const regularFare = getFareBetween(
   121→   121→      pickupPoint.pointNumber,
   122→   122→      dropoffPoint.pointNumber,
   123→   123→      false
   124→   124→    );
   125→   125→    const discountedFare = getFareBetween(
   126→   126→      pickupPoint.pointNumber,
   127→   127→      dropoffPoint.pointNumber,
   128→   128→      true
   129→   129→    );
   130→   130→    const finalFare = isDiscounted ? discountedFare : regularFare;
   131→   131→    const discountAmount = regularFare - discountedFare;
   132→   132→    const succeedingCount = Math.max(
   133→   133→      0,
   134→   134→      barangaysTraveled - FARE_CONFIG.BASE_BARANGAY_COUNT
   135→   135→    );
   136→   136→
   137→   137→    return {
   138→   138→      barangaysTraveled,
   139→   139→      regularFare,
   140→   140→      discountedFare,
   141→   141→      finalFare,
   142→   142→      hasDiscount: isDiscounted,
   143→   143→      discountAmount,
   144→   144→      succeedingCount,
   145→   145→      baseBarangayCount: FARE_CONFIG.BASE_BARANGAY_COUNT,
